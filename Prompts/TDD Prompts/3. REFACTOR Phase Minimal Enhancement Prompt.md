            'test_type': self.test_type,

        }

            'cache_hits': 0,

        }

        """
        Store basic test information with enhanced validation
        
        Returns:
            Tuple of (success: bool, message: str)
        """
        try:
            # Validate required fields
            required_fields = ['test_id', 'position_context', 'test_type']
            missing_fields = [field for field in required_fields if field not in test_data]
            
            if missing_fields:
                return False, f"Missing required fields: {', '.join(missing_fields)}"
            
            # Create structured metadata
            metadata = TestMetadata(
                test_id=test_data['test_id'],
                position_context=test_data['position_context'],
                test_type=test_data['test_type'],
                created_at=test_data.get('created_at', datetime.now().isoformat()),
                stored_at=datetime.now().isoformat(),
                mobile_context=test_data.get('mobile_context')
            )

                json.dump(metadata.to_dict(), f, indent=2)

                self._position_cache[metadata.position_context].append(metadata.to_dict())
            
            # Update metrics
            self.metrics['store_operations'] += 1
            
            self.logger.info(f"Successfully stored test {metadata.test_id}")
            return True, "Test stored successfully"
            
        except Exception as e:
            self.logger.error(f"Error storing test data: {e}")
            return False, f"Storage error: {str(e)}"

        """
        Retrieve tests by contextual position with caching
        
        Returns:
            Tuple of (tests: List, status_message: str)

            # Check cache first
            if position_context in self._position_cache:

                return self._position_cache[position_context], "Retrieved from cache"

            tests = []

                        tests.append(test_data)

                    continue
                    
            # Update cache
            self._position_cache[position_context] = tests
            self.metrics['retrieve_operations'] += 1
            
            return tests, f"Retrieved {len(tests)} tests"
            
        except Exception as e:
            self.logger.error(f"Error retrieving tests: {e}")
            return [], f"Retrieval error: {str(e)}"

import json

from typing import Dict, Any, List, Literal, Optional, Tuple

    """Enumerated component status types with validation"""
    READY = "ready"
    NOT_READY = "not_ready" 
    IN_PROGRESS = "in_progress"
    BLOCKED = "blocked"

    status: ComponentStatus

class ComponentStatusRepository:
    """Enhanced repository for component development status tracking"""
    
    def __init__(self, storage_path: str = "./data/components", enable_metrics: bool = True):
        """Initialize with optional metrics collection"""
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.enable_metrics = enable_metrics
        
        # Setup logging
        self.logger = logging.getLogger(__name__)
        
        # Status transition metrics
        self.metrics = {
            'status_changes': {},
            'readiness_checks': 0,
            'batch_operations': 0,
            'error_count': 0
        } if enable_metrics else {}
    
    def store_status(self, status_data: Dict[str, Any]) -> Tuple[bool, str]:
        """
        Store component development status with enhanced validation
        
        Returns:
            Tuple of (success: bool, message: str)
        """
        try:
            # Validate required fields
            required_fields = ['component_id', 'status', 'position']
            missing_fields = [field for field in required_fields if field not in status_data]
            
            if missing_fields:
                return False, f"Missing required fields: {', '.join(missing_fields)}"
            
            # Validate status enum
            try:
                status_enum = ComponentStatus(status_data['status'])
            except ValueError:
                valid_statuses = [s.value for s in ComponentStatus]
                return False, f"Invalid status. Must be one of: {', '.join(valid_statuses)}"
            
            # Create structured data
            component_data = ComponentStatusData(
                component_id=status_data['component_id'],
                status=status_enum,
                position=status_data['position'],
                updated_at=status_data.get('updated_at', datetime.now().isoformat()),
                last_updated=datetime.now().isoformat(),
                metadata=status_data.get('metadata', {})
            )
            
            # Check for status transition
            previous_status = self._get_previous_status(component_data.component_id)
            if previous_status and self.enable_metrics:
                transition = f"{previous_status} -> {status_enum.value}"
                self.metrics['status_changes'][transition] = self.metrics['status_changes'].get(transition, 0) + 1
            
            # Store to file
            filename = f"{component_data.component_id}_status.json"
            filepath = self.storage_path / filename
            
            with open(filepath, 'w', encoding='utf-8') as f:
                json.dump(component_data.to_dict(), f, indent=2)
            
            self.logger.info(f"Status updated for component {component_data.component_id}: {status_enum.value}")
            return True, f"Status updated successfully to {status_enum.value}"
            
        except Exception as e:
            if self.enable_metrics:
                self.metrics['error_count'] += 1
            self.logger.error(f"Error storing component status: {e}")
            return False, f"Storage error: {str(e)}"
    
    def check_readiness(self, component_id: str) -> Tuple[ComponentStatus, Dict[str, Any]]:
        """
        Enhanced readiness checking with detailed information
        
        Returns:
            Tuple of (status: ComponentStatus, info: Dict)
        """
        try:
            if self.enable_metrics:
                self.metrics['readiness_checks'] += 1
                
            filename = f"{component_id}_status.json"
            filepath = self.storage_path / filename
            
            if not filepath.exists():
                return ComponentStatus.NOT_READY, {
                    'message': 'Component not found',
                    'checked_at': datetime.now().isoformat()
                }
                
            # Load status data
            with open(filepath, 'r', encoding='utf-8') as f:
                status_data = json.load(f)
                
            status = ComponentStatus(status_data.get('status', 'not_ready'))
            
            # Calculate time since last update
            last_updated = status_data.get('last_updated')
            time_since_update = None
            if last_updated:
                try:
                    last_update_time = datetime.fromisoformat(last_updated)
                    time_since_update = (datetime.now() - last_update_time).total_seconds()
                except ValueError:
                    pass
            
            info = {
                'status': status.value,
                'position': status_data.get('position', 'unknown'),
                'last_updated': last_updated,
                'time_since_update_seconds': time_since_update,
                'metadata': status_data.get('metadata', {}),
                'checked_at': datetime.now().isoformat()
            }
            
            return status, info
            
        except Exception as e:
            if self.enable_metrics:
                self.metrics['error_count'] += 1
            self.logger.error(f"Error checking component readiness: {e}")
            return ComponentStatus.NOT_READY, {
                'error': str(e),
                'checked_at': datetime.now().isoformat()
            }
    
    def batch_update_statuses(self, updates: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Batch update multiple component statuses"""
        results = {
            'successful': 0,
            'failed': 0,
            'errors': [],
            'updated_components': []
        }
        
        try:
            if self.enable_metrics:
                self.metrics['batch_operations'] += 1
                
            for update in updates:
                success, message = self.store_status(update)
                if success:
                    results['successful'] += 1
                    results['updated_components'].append(update.get('component_id', 'unknown'))
                else:
                    results['failed'] += 1
                    results['errors'].append({
                        'component_id': update.get('component_id', 'unknown'),
                        'error': message
                    })
                    
            return results
            
        except Exception as e:
            self.logger.error(f"Batch update error: {e}")
            results['errors'].append({'batch_error': str(e)})
            return results

# Enhanced WorkflowRepository with state machines and PROJECT-002 integration
import json
import logging
from typing import Dict, Any, List, Literal, Optional, Tuple, Callable
from datetime import datetime, timedelta
from pathlib import Path
from dataclasses import dataclass
from enum import Enum
import asyncio

class ProgressionState(Enum):
    """Enhanced progression states with intermediate states"""
    READY = "ready"
    INITIALIZING = "initializing" 
    IN_PROGRESS = "in_progress"
    VALIDATING = "validating"
    COMPLETED = "completed"
    BLOCKED = "blocked"
    ROLLBACK = "rollback"

@dataclass
class WorkflowTransition:
    """Workflow state transition with metadata"""
    from_state: ProgressionState
    to_state: ProgressionState
    timestamp: str
    trigger: str
    metadata: Dict[str, Any]
    
class WorkflowStateMachine:
    """State machine for PROJECT-002 workflow progression"""
    
    # Valid state transitions
    TRANSITIONS = {
        ProgressionState.READY: [ProgressionState.INITIALIZING],
        ProgressionState.INITIALIZING: [ProgressionState.IN_PROGRESS, ProgressionState.BLOCKED],
        ProgressionState.IN_PROGRESS: [ProgressionState.VALIDATING, ProgressionState.BLOCKED, ProgressionState.ROLLBACK],
        ProgressionState.VALIDATING: [ProgressionState.COMPLETED, ProgressionState.IN_PROGRESS, ProgressionState.BLOCKED],
        ProgressionState.COMPLETED: [ProgressionState.READY],  # For next layer
        ProgressionState.BLOCKED: [ProgressionState.IN_PROGRESS, ProgressionState.ROLLBACK],
        ProgressionState.ROLLBACK: [ProgressionState.READY, ProgressionState.IN_PROGRESS]
    }
    
    def __init__(self, current_state: ProgressionState = ProgressionState.READY):
        self.current_state = current_state
        self.transition_history: List[WorkflowTransition] = []
    
    def can_transition_to(self, target_state: ProgressionState) -> bool:
        """Check if transition to target state is valid"""
        return target_state in self.TRANSITIONS.get(self.current_state, [])
    
    def transition_to(self, target_state: ProgressionState, trigger: str, metadata: Dict[str, Any] = None) -> bool:
        """Execute state transition with validation"""
        if not self.can_transition_to(target_state):
            return False
            
        # Record transition
        transition = WorkflowTransition(
            from_state=self.current_state,
            to_state=target_state,
            timestamp=datetime.now().isoformat(),
            trigger=trigger,
            metadata=metadata or {}
        )
        
        self.transition_history.append(transition)
        self.current_state = target_state
        return True

class WorkflowRepository:
    """Enhanced repository for PROJECT-002 workflow progression with state machines"""
    
    def __init__(self, storage_path: str = "./data/workflows", notification_callbacks: List[Callable] = None):
        """Initialize with optional notification system"""
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        
        # Notification system for PROJECT-002 integration
        self.notification_callbacks = notification_callbacks or []
        
        # Workflow state machines by workflow_id
        self._state_machines: Dict[str, WorkflowStateMachine] = {}
        
        # Performance and monitoring
        self.metrics = {
            'workflows_created': 0,
            'progressions_triggered': 0,
            'state_transitions': 0,
            'notifications_sent': 0,
            'rollbacks_executed': 0
        }
        
        # Setup logging
        self.logger = logging.getLogger(__name__)
        
        # Load existing workflows
        self._load_existing_workflows()
    
    def store_workflow_data(self, workflow_data: Dict[str, Any]) -> Tuple[bool, str]:
        """
        Enhanced workflow data storage with state machine integration
        
        Returns:
            Tuple of (success: bool, message: str)
        """
        try:
            # Validate required fields
            required_fields = ['workflow_id', 'current_layer', 'progression_status']
            missing_fields = [field for field in required_fields if field not in workflow_data]
            
            if missing_fields:
                return False, f"Missing required fields: {', '.join(missing_fields)}"
            
            workflow_id = workflow_data['workflow_id']
            
            # Validate progression status
            try:
                target_state = ProgressionState(workflow_data['progression_status'])
            except ValueError:
                valid_states = [s.value for s in ProgressionState]
                return False, f"Invalid status. Must be one of: {', '.join(valid_states)}"
            
            # Get or create state machine
            if workflow_id not in self._state_machines:
                self._state_machines[workflow_id] = WorkflowStateMachine()
                self.metrics['workflows_created'] += 1
            
            state_machine = self._state_machines[workflow_id]
            
            # Attempt state transition
            if state_machine.current_state != target_state:
                transition_success = state_machine.transition_to(
                    target_state,
                    trigger="manual_update",
                    metadata={
                        'layer': workflow_data['current_layer'],
                        'next_layer': workflow_data.get('next_layer'),
                        'updated_by': workflow_data.get('updated_by', 'system')
                    }
                )
                
                if not transition_success:
                    return False, f"Invalid state transition from {state_machine.current_state.value} to {target_state.value}"
                
                self.metrics['state_transitions'] += 1
            
            # Enhance workflow data with state machine info
            enhanced_data = {
                **workflow_data,
                'last_updated': datetime.now().isoformat(),
                'current_state': state_machine.current_state.value,
                'transition_history': [
                    {
                        'from_state': t.from_state.value,
                        'to_state': t.to_state.value,
                        'timestamp': t.timestamp,
                        'trigger': t.trigger,
                        'metadata': t.metadata
                    } for t in state_machine.transition_history[-5:]  # Last 5 transitions
                ],
                'valid_next_states': [s.value for s in WorkflowStateMachine.TRANSITIONS.get(state_machine.current_state, [])]
            }
            
            # Store to file
            filename = f"{workflow_id}.json"
            filepath = self.storage_path / filename
            
            with open(filepath, 'w', encoding='utf-8') as f:
                json.dump(enhanced_data, f, indent=2)
            
            # Send notifications if state changed
            if state_machine.transition_history:
                self._send_notifications(workflow_id, state_machine.transition_history[-1])
            
            self.logger.info(f"Workflow {workflow_id} updated to state {target_state.value}")
            return True, f"Workflow updated successfully to {target_state.value}"
            
        except Exception as e:
            self.logger.error(f"Error storing workflow data: {e}")
            return False, f"Storage error: {str(e)}"
    
    async def trigger_automatic_progression(self, from_layer: str, to_layer: str, validation_checks: List[Callable] = None) -> Tuple[bool, str]:
        """
        Advanced automatic progression with validation pipeline
        
        Returns:
            Tuple of (success: bool, message: str)
        """
        try:
            self.metrics['progressions_triggered'] += 1
            
            # Run validation checks if provided
            if validation_checks:
                for check in validation_checks:
                    check_result = await check(from_layer, to_layer) if asyncio.iscoroutinefunction(check) else check(from_layer, to_layer)
                    if not check_result:

def _validate_and_normalize_config(self, config: Dict[str, Any]) -> Dict[str, Any]:
    """Validate and normalize configuration parameters with intelligent defaults"""

        ),
        'max_recovery_time_s': self._validate_range(

            'max_recovery_time_s must be between 1-30 seconds'
        ),
        'max_mobile_package_kb': self._validate_range(
            perf_limits.get('max_mobile_package_kb', 50), 10, 500,
            'max_mobile_package_kb must be between 10-500KB'
        )
    }
    
    # Validate display formats
    display_formats = config.get('display_formats', {})
    normalized_config['display_formats'] = {
        'timestamp_format': display_formats.get('timestamp_format', '%Y-%m-%d %H:%M:%S'),
        'percentage_precision': self._validate_range(
            display_formats.get('percentage_precision', 1), 0, 3,
            'percentage_precision must be between 0-3 decimal places'
        ),
        'file_size_units': self._validate_choice(
            display_formats.get('file_size_units', 'bytes'),
            ['bytes', 'KB', 'MB'], 'file_size_units'
        )
    }
    
    # Validate compliance thresholds
    compliance_thresholds = config.get('compliance_thresholds', {})
    excellent = compliance_thresholds.get('excellent_score', 90.0)
    good = compliance_thresholds.get('good_score', 80.0)
    acceptable = compliance_thresholds.get('acceptable_score', 70.0)
    
    # Ensure logical threshold ordering
    if not (acceptable < good < excellent):
        raise ValueError(
            f"Compliance thresholds must be ordered: acceptable ({acceptable}) < "
            f"good ({good}) < excellent ({excellent})"
        )
    
    normalized_config['compliance_thresholds'] = {
        'excellent_score': excellent,
        'good_score': good,
        'acceptable_score': acceptable
    }
    
    return normalized_config

def _validate_range(self, value: Any, min_val: float, max_val: float, error_msg: str) -> float:
    """Validate numeric value is within specified range"""
    if not isinstance(value, (int, float)):
        raise ValueError(f"{error_msg}: must be numeric, got {type(value)}")
    
    if not min_val <= value <= max_val:
        raise ValueError(f"{error_msg}: got {value}")
    
    return float(value)

def _validate_choice(self, value: Any, choices: List[str], param_name: str) -> str:
    """Validate value is one of allowed choices"""
    if value not in choices:
        raise ValueError(f"{param_name} must be one of {choices}, got '{value}'")
    
    return str(value)
