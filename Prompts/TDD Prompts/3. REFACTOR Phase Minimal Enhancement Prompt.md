class DisplayRenderer:
    """Reusable display rendering engine for consistent formatting"""
    
    @staticmethod
    def format_status_line(stage: str, status: str, details: Dict[str, Any] = None) -> str:
        """Format consistent status lines across all display methods"""
        line = f"{stage}: {status}"
        if details and status == 'failed':
            line += f"\n  error: {details.get('error_message', 'Unknown error')}"
            line += f"\n  retry_count: {details.get('retry_count', 0)}"
        return line
    
    @staticmethod
    def format_compliance_section(title: str, data: Dict[str, Any]) -> List[str]:
        """Format compliance data sections consistently"""
        lines = [f"{title}:"]
        for key, value in data.items():
            if isinstance(value, dict):
                score = value.get('score', 0)
                status = value.get('status', 'unknown')
                violations = value.get('violations', 0)
                lines.append(f"  {key}: {score}% - {status} - violations: {violations}")
            else:
                lines.append(f"  {key}: {value}")
        return lines
```

class InvalidComplianceDataError(DisplayValidationError):

def _validate_evidence_event(self, evidence_event: Dict[str, Any]) -> None:
    """Validate evidence event structure and required fields."""
    required_fields = ['stage', 'evidence_type', 'status']
    missing_fields = [field for field in required_fields if field not in evidence_event]
    
    if missing_fields:
        raise InvalidEvidenceEventError(
            f"Missing required fields: {missing_fields}. "
            f"Required: {required_fields}"
        )
    
    valid_stages = ['red_stage', 'green_stage', 'refactor_stage']
    if evidence_event['stage'] not in valid_stages:
        raise InvalidEvidenceEventError(
            f"Invalid stage '{evidence_event['stage']}'. "
            f"Must be one of: {valid_stages}"
        )
    
    valid_statuses = ['collecting', 'completed', 'failed', 'pending']
    if evidence_event['status'] not in valid_statuses:
        raise InvalidEvidenceEventError(
            f"Invalid status '{evidence_event['status']}'. "
            f"Must be one of: {valid_statuses}"
        )

def _validate_compliance_data(self, compliance_data: Dict[str, Any]) -> None:
    """Validate compliance data structure and value ranges."""
    if 'overall_compliance_score' not in compliance_data:
        raise InvalidComplianceDataError("Missing 'overall_compliance_score' field")
    
    score = compliance_data['overall_compliance_score']
    if not isinstance(score, (int, float)) or not 0 <= score <= 100:
        raise InvalidComplianceDataError(
            f"overall_compliance_score must be numeric between 0-100, got {score}"
        )

                                       size_bytes: int, 
                                       line_count: int, 
                                       complexity: float) -> str:
        """Cached artifact detail formatting for repeated requests"""
        return f"""Artifact: {name}
Size: {size_bytes:,} bytes
{line_count} lines
Complexity: {complexity}"""
    
    def _get_cached_chart(self, chart_type: str, data_hash: str) -> Optional[Dict[str, Any]]:
        """Retrieve cached chart if available and valid"""
        cache_key = f"{chart_type}_{data_hash}"
        cached_entry = self._chart_cache.get(cache_key)
        
        if cached_entry:
            # Check if cache is still valid (5 minute expiry)
            cache_age = time.time() - cached_entry['created_at']
            if cache_age < 300:  # 5 minutes
                self.logger.debug(f"Using cached chart: {cache_key}")
                return cached_entry['chart_data']
        
        return None
    
    def _cache_chart(self, chart_type: str, data_hash: str, chart_data: Dict[str, Any]) -> None:
        """Cache generated chart data"""
        cache_key = f"{chart_type}_{data_hash}"
        self._chart_cache[cache_key] = {
            'chart_data': chart_data,
            'created_at': time.time()
        }
        
        # Limit cache size
        if len(self._chart_cache) > 50:
            oldest_key = min(self._chart_cache.keys(), 
                           key=lambda k: self._chart_cache[k]['created_at'])
            del self._chart_cache[oldest_key]

def _update_performance_metrics(self, operation_type: str, duration_ms: float) -> None:
    """Update internal performance tracking metrics"""
    if operation_type == 'update':
        self.performance_metrics['updates_processed'] += 1
        # Update rolling average response time
        current_avg = self.performance_metrics['average_response_time_ms']
        total_updates = self.performance_metrics['updates_processed']
        new_avg = ((current_avg * (total_updates - 1)) + duration_ms) / total_updates
        self.performance_metrics['average_response_time_ms'] = new_avg
        
    elif operation_type == 'report':
        self.performance_metrics['reports_generated'] += 1
        
    elif operation_type == 'recovery':
        self.performance_metrics['errors_recovered'] += 1

def get_performance_report(self) -> Dict[str, Any]:
    """Generate performance metrics report for monitoring"""
    return {
        'performance_summary': self.performance_metrics.copy(),
        'performance_status': {
            'response_time_acceptable': self.performance_metrics['average_response_time_ms'] < 100,
            'error_rate_low': self.performance_metrics['errors_recovered'] < 5,
            'throughput_adequate': self.performance_metrics['updates_processed'] > 0
        },
        'recommendations': self._generate_performance_recommendations()
    }

def _generate_performance_recommendations(self) -> List[str]:
    """Generate performance optimization recommendations"""
    recommendations = []
    
    if self.performance_metrics['average_response_time_ms'] > 50:
        recommendations.append("Consider enabling caching for repeated operations")
    
    if self.performance_metrics['errors_recovered'] > 10:
        recommendations.append("Review error handling - high recovery rate detected")
    
    if len(self._chart_cache) > 40:
        recommendations.append("Chart cache approaching limit - consider cleanup")
    
    return recommendations
```

def _validate_and_normalize_config(self, config: Dict[str, Any]) -> Dict[str, Any]:
    """Validate and normalize configuration parameters with intelligent defaults"""
    normalized_config = {}
    
    # Validate performance limits
    perf_limits = config.get('performance_limits', {})
    normalized_config['performance_limits'] = {
        'max_update_response_ms': self._validate_range(
            perf_limits.get('max_update_response_ms', 100), 1, 1000,
            'max_update_response_ms must be between 1-1000ms'
        ),
        'max_recovery_time_s': self._validate_range(
            perf_limits.get('max_recovery_time_s', 2), 1, 30,
            'max_recovery_time_s must be between 1-30 seconds'
        ),
        'max_mobile_package_kb': self._validate_range(
            perf_limits.get('max_mobile_package_kb', 50), 10, 500,
            'max_mobile_package_kb must be between 10-500KB'
        )
    }
    
    # Validate display formats
    display_formats = config.get('display_formats', {})
    normalized_config['display_formats'] = {
        'timestamp_format': display_formats.get('timestamp_format', '%Y-%m-%d %H:%M:%S'),
        'percentage_precision': self._validate_range(
            display_formats.get('percentage_precision', 1), 0, 3,
            'percentage_precision must be between 0-3 decimal places'
        ),
        'file_size_units': self._validate_choice(
            display_formats.get('file_size_units', 'bytes'),
            ['bytes', 'KB', 'MB'], 'file_size_units'
        )
    }
    
    # Validate compliance thresholds
    compliance_thresholds = config.get('compliance_thresholds', {})
    excellent = compliance_thresholds.get('excellent_score', 90.0)
    good = compliance_thresholds.get('good_score', 80.0)
    acceptable = compliance_thresholds.get('acceptable_score', 70.0)
    
    # Ensure logical threshold ordering
    if not (acceptable < good < excellent):
        raise ValueError(
            f"Compliance thresholds must be ordered: acceptable ({acceptable}) < "
            f"good ({good}) < excellent ({excellent})"
        )
    
    normalized_config['compliance_thresholds'] = {
        'excellent_score': excellent,
        'good_score': good,
        'acceptable_score': acceptable
    }
    
    return normalized_config

def _validate_range(self, value: Any, min_val: float, max_val: float, error_msg: str) -> float:
    """Validate numeric value is within specified range"""
    if not isinstance(value, (int, float)):
        raise ValueError(f"{error_msg}: must be numeric, got {type(value)}")
    
    if not min_val <= value <= max_val:
        raise ValueError(f"{error_msg}: got {value}")
    
    return float(value)

def _validate_choice(self, value: Any, choices: List[str], param_name: str) -> str:
    """Validate value is one of allowed choices"""
    if value not in choices:
        raise ValueError(f"{param_name} must be one of {choices}, got '{value}'")
    
    return str(value)
